### Solution 1: Error Boundary Component

**File:** `src/components/common/ErrorBoundary.tsx`

```typescript
import { Component, ErrorInfo, ReactNode } from 'react';
import { AlertTriangle, RefreshCw, Home } from 'lucide-react';
import { Button } from '../ui/button';

interface Props {
  children: ReactNode;
  fallback?: ReactNode;
  onError?: (error: Error, errorInfo: ErrorInfo) => void;
}

interface State {
  hasError: boolean;
  error: Error | null;
}

export class ErrorBoundary extends Component<Props, State> {
  public state: State = {
    hasError: false,
    error: null,
  };

  public static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  public componentDidCatch(error: Error, errorInfo: ErrorInfo) {
    console.error('[ErrorBoundary] Caught error:', error, errorInfo);
    
    // Call custom error handler if provided
    this.props.onError?.(error, errorInfo);
    
    // TODO: Send to error tracking service (Sentry, LogRocket, etc.)
    // sendToSentry(error, errorInfo);
  }

  public handleReload = () => {
    window.location.reload();
  };

  public handleGoHome = () => {
    window.location.href = '/';
  };

  public render() {
    if (this.state.hasError) {
      if (this.props.fallback) {
        return this.props.fallback;
      }

      return (
        <div className="min-h-[400px] flex items-center justify-center bg-bg-section rounded-lg">
          <div className="text-center p-8 max-w-md">
            <div className="inline-flex items-center justify-center w-16 h-16 rounded-full bg-error/10 mb-4">
              <AlertTriangle className="w-8 h-8 text-error" />
            </div>
            
            <h2 className="text-2xl font-display font-semibold mb-2">
              Oops! Something went wrong
            </h2>
            
            <p className="text-text-secondary mb-6">
              We're sorry for the inconvenience. Please try again.
            </p>
            
            {this.state.error && (
              <details className="mb-6 text-left">
                <summary className="cursor-pointer text-sm text-text-muted hover:text-text-secondary">
                  Error details
                </summary>
                <pre className="mt-2 p-4 bg-bg-card rounded text-xs text-error overflow-auto">
                  {this.state.error.message}
                </pre>
              </details>
            )}
            
            <div className="flex gap-3 justify-center">
              <Button
                onClick={this.handleReload}
                className="gap-2"
              >
                <RefreshCw className="w-4 h-4" />
                Reload Page
              </Button>
              
              <Button
                onClick={this.handleGoHome}
                variant="outline"
                className="gap-2"
              >
                <Home className="w-4 h-4" />
                Go Home
              </Button>
            </div>
          </div>
        </div>
      );
    }

    return this.props.children;
  }
}
```

### Solution 2: Error Handler Hook

**File:** `src/hooks/useErrorHandler.ts`

```typescript
import { useCallback } from 'react';
import { toast } from 'sonner';

interface ErrorOptions {
  title?: string;
  message?: string;
  showToast?: boolean;
  logError?: boolean;
  retry?: () => void;
}

interface ApiError {
  message: string;
  code?: string;
}

/**
 * Error Handler Hook
 * 
 * Provides consistent error handling across the app
 */
export const useErrorHandler = () => {
  /**
   * Handle generic errors
   */
  const handleError = useCallback((error: unknown, options?: ErrorOptions) => {
    const {
      title = 'Error',
      message,
      showToast = true,
      logError = true,
    } = options || {};

    // Log error
    if (logError) {
      console.error('[Error Handler]:', error);
    }

    // Extract error message
    const errorMessage = message || (
      error instanceof Error 
        ? error.message 
        : 'An unexpected error occurred'
    );

    // Show toast
    if (showToast) {
      toast.error(title, {
        description: errorMessage,
        duration: 5000,
        action: options.retry ? {
          label: 'Retry',
          onClick: options.retry,
        } : undefined,
      });
    }

    // TODO: Send to error tracking
    // sendToSentry(error);
  }, []);

  /**
   * Handle API errors specifically
   */
  const handleApiError = useCallback(async (
    response: Response, 
    options?: ErrorOptions
  ) => {
    try {
      const errorData: ApiError = await response.json();
      handleError(errorData, {
        ...options,
        message: errorData.message || 'API request failed',
      });
    } catch {
      handleError(new Error('Network error'), options);
    }
  }, [handleError]);

  /**
   * Handle validation errors
   */
  const handleValidationError = useCallback((
    errors: { field: string; message: string }[]
  ) => {
    errors.forEach(error => {
      toast.error('Validation Error', {
        description: `${error.field}: ${error.message}`,
        duration: 5000,
      });
    });
  }, []);

  return { 
    handleError, 
    handleApiError, 
    handleValidationError 
  };
};
```

import { useErrorHandler } from '../hooks/useErrorHandler';

const Dashboard = () => {
  const { handleError } = useErrorHandler();
  const [isLoading, setIsLoading] = useState(false);

  const loadDashboardData = async () => {
    try {
      setIsLoading(true);
      
      const competitionService = getService('competition');
      const notificationService = getService('notification');
      
      const [competitions, notifications] = await Promise.all([
        competitionService.getAll(),
        notificationService.getAll(),
      ]);
      
      setCompetitions(competitions);
      setNotifications(notifications);
      
    } catch (error) {
      handleError(error, {
        title: 'Failed to load data',
        message: 'Please check your connection and try again.',
        retry: loadDashboardData,
      });
    } finally {
      setIsLoading(false);
    }
  };

  // ... rest of component
};

```typescript
import { Inbox, Trophy, Users, FileX } from 'lucide-react';
import { Button } from '../ui/button';

interface EmptyStateProps {
  type: 'competitions' | 'teams' | 'submissions' | 'notifications' | 'general';
  onAction?: () => void;
  actionLabel?: string;
}

export const EmptyState = ({ type, onAction, actionLabel }: EmptyStateProps) => {
  const configs = {
    competitions: {
      icon: Trophy,
      title: 'No competitions yet',
      description: 'Start your journey by registering for a competition.',
    },
    teams: {
      icon: Users,
      title: 'No teams yet',
      description: 'Create a team or join an existing one.',
    },
    submissions: {
      icon: FileX,
      title: 'No submissions yet',
      description: 'Submit your work when you\'re ready.',
    },
    notifications: {
      icon: Inbox,
      title: 'No notifications',
      description: 'You\'re all caught up!',
    },
    general: {
      icon: Inbox,
      title: 'Nothing here',
      description: 'There\'s nothing to show here yet.',
    },
  };

  const config = configs[type] || configs.general;
  const Icon = config.icon;

  return (
    <div className="min-h-[300px] flex items-center justify-center">
      <div className="text-center p-8">
        <div className="inline-flex items-center justify-center w-16 h-16 rounded-full bg-bg-section mb-4">
          <Icon className="w-8 h-8 text-text-muted" />
        </div>
        
        <h3 className="text-lg font-semibold mb-2">
          {config.title}
        </h3>
        
        <p className="text-text-secondary mb-6">
          {config.description}
        </p>
        
        {onAction && actionLabel && (
          <Button onClick={onAction}>
            {actionLabel}
          </Button>
        )}
      </div>
    </div>
  );
};
```
