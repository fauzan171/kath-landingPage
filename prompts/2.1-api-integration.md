/**
 * Environment Configuration
 * 
 * Controls switching between mock and real API
 * Based on environment variables
 */

export interface EnvironmentConfig {
  apiUrl: string;
  useMockData: boolean;
  environment: 'development' | 'staging' | 'production';
  debug: boolean;
}

/**
 * Get current environment configuration
 */
const getEnvironment = (): EnvironmentConfig => {
  const isDev = import.meta.env.DEV;
  const isProd = import.meta.env.PROD;
  
  return {
    apiUrl: import.meta.env.VITE_API_URL || 'http://localhost:3001/api',
    useMockData: import.meta.env.VITE_USE_MOCK_DATA === 'true',
    environment: isProd ? 'production' : isDev ? 'development' : 'staging',
    debug: isDev,
  };
};

export const env = getEnvironment();

/**
 * Log environment info in development
 */
if (env.debug) {
  console.log('[Environment]', {
    apiUrl: env.apiUrl,
    useMockData: env.useMockData,
    environment: env.environment,
  });
}

### Task 2: Create Service Factory

**File:** `src/services/service.factory.ts`

```typescript
/**
 * Service Factory
 * 
 * Provides unified interface for switching between
 * mock and real API implementations
 */

import { env } from '../config/environment';

export type ServiceType = 
  | 'competition' 
  | 'portfolio' 
  | 'news' 
  | 'auth' 
  | 'user'
  | 'team'
  | 'submission'
  | 'notification';

/**
 * Service Factory Class
 * 
 * Singleton pattern for service instances
 */
export class ServiceFactory {
  private static instance: ServiceFactory;
  
  private constructor() {}
  
  static getInstance(): ServiceFactory {
    if (!ServiceFactory.instance) {
      ServiceFactory.instance = new ServiceFactory();
    }
    return ServiceFactory.instance;
  }
  
  /**
   * Get service implementation based on environment
   */
  getService<T extends ServiceType>(type: T) {
    const useMock = env.useMockData;
    
    switch (type) {
      case 'competition':
        return useMock ? mockCompetitionService : apiCompetitionService;
      case 'portfolio':
        return useMock ? mockPortfolioService : apiPortfolioService;
      case 'news':
        return useMock ? mockNewsService : apiNewsService;
      case 'auth':
        return useMock ? mockAuthService : apiAuthService;
      case 'user':
        return useMock ? mockUserService : apiUserService;
      case 'team':
        return useMock ? mockTeamService : apiTeamService;
      case 'submission':
        return useMock ? mockSubmissionService : apiSubmissionService;
      case 'notification':
        return useMock ? mockNotificationService : apiNotificationService;
      default:
        throw new Error(`Unknown service type: ${type}`);
    }
  }
}

// Service interfaces
export interface CompetitionService {
  getAll: (filters?: any) => Promise<any[]>;
  getById: (id: string) => Promise<any>;
  register: (id: string, data: any) => Promise<{ success: boolean; message: string }>;
}

// Service instances (will be imported from respective modules)
import { 
  competitionService as mockCompetitionService,
  portfolioService as mockPortfolioService,
  newsService as mockNewsService,
  authService as mockAuthService,
  userService as mockUserService,
  teamService as mockTeamService,
  submissionService as mockSubmissionService,
  notificationService as mockNotificationService,
} from './mockData';

import {
  competitionService as apiCompetitionService,
  portfolioService as apiPortfolioService,
  newsService as apiNewsService,
  authService as apiAuthService,
  userService as apiUserService,
  teamService as apiTeamService,
  submissionService as apiSubmissionService,
  notificationService as apiNotificationService,
} from './api.services';

/**
 * Convenience function for getting services
 */
export function getService<T extends ServiceType>(type: T) {
  return ServiceFactory.getInstance().getService(type);
}
```

import { api } from './api';
import type { Competition, Team, Submission, Notification } from './mockData';

// Competition API Service
export const competitionService = {
  async getAll(filters?: {
    status?: 'registered' | 'in_progress' | 'finished' | 'upcoming';
    category?: string;
  }): Promise<Competition[]> {
    const params: Record<string, any> = {};
    if (filters?.status) params.status = filters.status;
    if (filters?.category) params.category = filters.category;
    
    const response = await api.get<Competition[]>('/competitions', params);
    return response.data;
  },

  async getById(id: string): Promise<Competition | undefined> {
    const response = await api.get<Competition>(`/competitions/${id}`);
    return response.data;
  },

  async register(
    competitionId: string, 
    userData: any
  ): Promise<{ success: boolean; message: string }> {
    const response = await api.post<{ message: string }>(
      `/competitions/${competitionId}/register`,
      userData
    );
    return { success: response.success, message: response.data.message };
  },

  async update(
    competitionId: string,
    updates: Partial<Competition>
  ): Promise<Competition> {
    const response = await api.put<Competition>(
      `/competitions/${competitionId}`,
      updates
    );
    return response.data;
  },
};

### Task 5: Update Components to Use Service Layer

// Before
import { getCompetitions, getDashboardStats, getNotifications } from '../services/mockData';

useEffect(() => {
  setNotifications(getNotifications());
  setCompetitions(getCompetitions());
  setStats(getDashboardStats());
}, []);

// After
import { getService } from '../services/service.factory';

useEffect(() => {
  const loadData = async () => {
    try {
      setIsLoading(true);
      
      const competitionService = getService('competition');
      const notificationService = getService('notification');
      
      const [competitions, notifications] = await Promise.all([
        competitionService.getAll(),
        notificationService.getAll(),
      ]);
      
      setCompetitions(competitions);
      setNotifications(notifications);
      setStats(getDashboardStats());
    } catch (error) {
      console.error('Failed to load dashboard data:', error);
      toast.error('Failed to load data');
    } finally {
      setIsLoading(false);
    }
  };
  
  loadData();
}, []);
